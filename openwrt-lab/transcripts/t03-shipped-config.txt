$ grep -A2 "ucidef_set_interface_lan()" lib/functions/uci-defaults.sh   (protocol defaults)
ucidef_set_interface_lan() {
	ucidef_set_interface "lan" device "$1" protocol "${2:-static}"
}
ucidef_set_interface_wan() {
	ucidef_set_interface "wan" device "$1" protocol "${2:-dhcp}"
}

(So on first boot a generic x86 machine gets: lan = eth0, static,
 192.168.1.1; wan = eth1 if present, dhcp; hostname OpenWrt.)
